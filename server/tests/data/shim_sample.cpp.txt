// Создано сервером сборки FPGA SIM. Не править.
#include "Vtop.h"
#include "fpgasim/engine_verilator.h"

namespace {

struct Shim final : fpgasim::VerilatorEngine<Vtop, Shim> {
    static constexpr fpgasim::PortDesc kPorts[] = {
        { "clk", fpgasim::PortDir::In, 1 },
        { "in8", fpgasim::PortDir::In, 8 },
        { "io1", fpgasim::PortDir::InOut, 1 },
        { "out1", fpgasim::PortDir::Out, 1 },
        { "led", fpgasim::PortDir::Out, 6 },
        { "out16", fpgasim::PortDir::Out, 16 },
        { "out32", fpgasim::PortDir::Out, 32 },
        { "out40", fpgasim::PortDir::Out, 40 },
        { "out70", fpgasim::PortDir::Out, 70 },
        { "bus-a", fpgasim::PortDir::In, 3 },
        { "in33", fpgasim::PortDir::In, 33 },
        { "in128", fpgasim::PortDir::In, 128 },
    };
    static constexpr const char* kTop = "top";

    // Общая часть обращается к портам через прослойку, поэтому модель создаётся здесь.
    Shim() { init(); }

    void* portData(int port) {
        switch (port) {
            case 0: return &top_->clk;
            case 1: return &top_->in8;
            case 2: return &top_->io1;
            case 3: return &top_->out1;
            case 4: return &top_->led;
            case 5: return &top_->out16;
            case 6: return &top_->out32;
            case 7: return &top_->out40;
            case 8: return &top_->out70;
            case 9: return &top_->bus__02Da;
            case 10: return &top_->in33;
            case 11: return &top_->in128;
        }
        return nullptr;
    }

    bool outputsChanged() {
        bool changed = false;
        if (FPGASIM_DIFFERS(top_->out1, prev_out1)) {
            FPGASIM_COPY(prev_out1, top_->out1);
            markChanged(3);
            changed = true;
        }
        if (FPGASIM_DIFFERS(top_->led, prev_led)) {
            FPGASIM_COPY(prev_led, top_->led);
            markChanged(4);
            changed = true;
        }
        if (FPGASIM_DIFFERS(top_->out16, prev_out16)) {
            FPGASIM_COPY(prev_out16, top_->out16);
            markChanged(5);
            changed = true;
        }
        if (FPGASIM_DIFFERS(top_->out32, prev_out32)) {
            FPGASIM_COPY(prev_out32, top_->out32);
            markChanged(6);
            changed = true;
        }
        if (FPGASIM_DIFFERS(top_->out40, prev_out40)) {
            FPGASIM_COPY(prev_out40, top_->out40);
            markChanged(7);
            changed = true;
        }
        if (FPGASIM_DIFFERS(top_->out70, prev_out70)) {
            FPGASIM_COPY(prev_out70, top_->out70);
            markChanged(8);
            changed = true;
        }
        return changed;
    }

    CData prev_out1{};
    CData prev_led{};
    SData prev_out16{};
    IData prev_out32{};
    QData prev_out40{};
    VlWide<3> prev_out70{};
};

}  // namespace

std::unique_ptr<fpgasim::Engine> fpgasim::createEngine() {
    return std::make_unique<Shim>();
}
